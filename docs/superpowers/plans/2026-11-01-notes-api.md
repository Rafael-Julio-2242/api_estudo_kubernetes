# Notes API Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a complete Notes REST API (TypeScript/Fastify 5/SQLite) with local and OIDC JWT auth, designed as a good Kubernetes citizen.

**Architecture:** Single Fastify 5 app built by a `buildApp(config, deps)` factory for testability; SQLite via better-sqlite3 with numbered SQL migrations; auth via jose (HS256/RS256 local or remote JWKS OIDC); graceful SIGTERM shutdown.

**Tech Stack:** Node.js 22 LTS, TypeScript strict ESM, Fastify 5, zod, better-sqlite3, jose, @node-rs/argon2, pino, Vitest 2, ESLint flat config + typescript-eslint, Prettier.

## Global Constraints

- TypeScript strict mode, ESM (`"type": "module"` in package.json)
- Node.js engine `>=22.0.0`; `.nvmrc` = `22`
- `npm` only; commit `package-lock.json`; pipelines use `npm ci`
- All npm scripts exit non-zero on failure; `npm run lint` uses `--max-warnings 0`
- IDs: `crypto.randomUUID()` (UUID v4)
- All timestamps: ISO 8601 UTC
- Body limit: 64 KB on all routes
- Error format: `{ "error": "<code>", "message": "<text>" }` — codes: `bad_request`, `unauthorized`, `forbidden`, `not_found`, `conflict`, `idp_unavailable`, `internal_error`
- Never log `JWT_SECRET`, private keys, passwords, or tokens
- `trustProxy: true` on the Fastify instance
- File imports always use `.js` extension (even for `.ts` source files)

```typescript
import { z } from 'zod'
import { readFileSync } from 'node:fs'
import { join } from 'node:path'

export class ConfigError extends Error {
  constructor(message: string) {
    super(message)
    this.name = 'ConfigError'
  }
}

function readEnvOrFile(name: string, env: Record<string, string | undefined>): string | undefined {
  const fileVar = env[`${name}_FILE`]
  if (fileVar) return readFileSync(fileVar, 'utf-8').trim()
  return env[name]
}

const schema = z
  .object({
    appEnv: z.string().default('local'),
    appVersion: z.string().default('0.0.0'),
    gitSha: z.string().default('unknown'),
    port: z.coerce.number().int().min(1).max(65535).default(8080),
    host: z.string().default('0.0.0.0'),
    logLevel: z.enum(['trace', 'debug', 'info', 'warn', 'error', 'fatal']).default('info'),
    databasePath: z.string().default('./data/notes.db'),
    shutdownDelaySeconds: z.coerce.number().int().min(0).default(0),
    authMode: z.enum(['local', 'oidc']).default('local'),
    jwtIssuer: z.string().default('notes-api'),
    jwtAudience: z.string().default('notes-api'),
    jwtAlg: z.enum(['HS256', 'RS256']).default('HS256'),
    jwtSecret: z.string().min(32).optional(),
    jwtPrivateKey: z.string().optional(),
    jwtPublicKey: z.string().optional(),
    jwtTtlSeconds: z.coerce.number().int().min(1).default(3600),
    oidcJwksUrl: z.string().url().optional(),
    seedAdminEmail: z.string().email().optional(),
    seedAdminPassword: z.string().optional(),
  })
  .superRefine((val, ctx) => {
    if (val.authMode === 'local') {
      if (val.jwtAlg === 'HS256' && !val.jwtSecret) {
        ctx.addIssue({ code: 'custom', message: 'JWT_SECRET required when AUTH_MODE=local and JWT_ALG=HS256 (min 32 chars)' })
      }
      if (val.jwtAlg === 'RS256' && (!val.jwtPrivateKey || !val.jwtPublicKey)) {
        ctx.addIssue({ code: 'custom', message: 'JWT_PRIVATE_KEY and JWT_PUBLIC_KEY required when JWT_ALG=RS256' })
      }
    }
  })

export type Config = z.infer<typeof schema>

function readPackageVersion(): string {
  try {
    const pkg = JSON.parse(readFileSync(join(process.cwd(), 'package.json'), 'utf-8')) as { version?: string }
    return pkg.version ?? '0.0.0'
  } catch {
    return '0.0.0'
  }
}

export function loadConfig(env: Record<string, string | undefined> = process.env): Config {
  const raw = {
    appEnv: env.APP_ENV,
    appVersion: env.APP_VERSION ?? readPackageVersion(),
    gitSha: env.GIT_SHA,
    port: env.PORT,
    host: env.HOST,
    logLevel: env.LOG_LEVEL,
    databasePath: env.DATABASE_PATH,
    shutdownDelaySeconds: env.SHUTDOWN_DELAY_SECONDS,
    authMode: env.AUTH_MODE,
    jwtIssuer: env.JWT_ISSUER,
    jwtAudience: env.JWT_AUDIENCE,
    jwtAlg: env.JWT_ALG,
    jwtSecret: readEnvOrFile('JWT_SECRET', env),
    jwtPrivateKey: readEnvOrFile('JWT_PRIVATE_KEY', env),
    jwtPublicKey: readEnvOrFile('JWT_PUBLIC_KEY', env),
    jwtTtlSeconds: env.JWT_TTL_SECONDS,
    oidcJwksUrl: env.OIDC_JWKS_URL,
    seedAdminEmail: env.SEED_ADMIN_EMAIL,
    seedAdminPassword: readEnvOrFile('SEED_ADMIN_PASSWORD', env),
  }

  const result = schema.safeParse(raw)
  if (!result.success) {
    const issues = result.error.issues.map((i) => `  - ${i.path.join('.')}: ${i.message}`).join('\n')
    throw new ConfigError(`Invalid configuration:\n${issues}`)
  }
  return result.data
}
```

```typescript
import type { FastifyError, FastifyReply, FastifyRequest } from 'fastify'

export type ErrorCode =
  | 'bad_request'
  | 'unauthorized'
  | 'forbidden'
  | 'not_found'
  | 'conflict'
  | 'idp_unavailable'
  | 'internal_error'

export class AppError extends Error {
  constructor(
    public readonly statusCode: number,
    public readonly code: ErrorCode,
    message: string,
  ) {
    super(message)
    this.name = 'AppError'
  }
}

export function errorHandler(
  error: FastifyError | Error,
  request: FastifyRequest,
  reply: FastifyReply,
): void {
  if (error instanceof AppError) {
    void reply.status(error.statusCode).send({ error: error.code, message: error.message })
    return
  }

  // Fastify validation errors (body/params)
  const fastifyErr = error as FastifyError
  if (fastifyErr.statusCode === 400 || fastifyErr.validation) {
    void reply.status(400).send({ error: 'bad_request', message: fastifyErr.message })
    return
  }

  request.log.error({ err: error }, 'Unhandled error')
  void reply.status(500).send({ error: 'internal_error', message: 'internal error' })
}
```

```typescript
import Database from 'better-sqlite3'
import { readdirSync, readFileSync } from 'node:fs'
import { join } from 'node:path'

export function runMigrations(db: Database.Database, migrationsDir: string): void {
  db.exec(`
    CREATE TABLE IF NOT EXISTS schema_migrations (
      filename TEXT PRIMARY KEY NOT NULL,
      applied_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ', 'now'))
    )
  `)

  const applied = new Set(
    (db.prepare('SELECT filename FROM schema_migrations').all() as { filename: string }[]).map(
      (r) => r.filename,
    ),
  )

  const files = readdirSync(migrationsDir)
    .filter((f) => f.endsWith('.sql'))
    .sort()

  for (const file of files) {
    if (applied.has(file)) continue
    const sql = readFileSync(join(migrationsDir, file), 'utf-8')
    db.transaction(() => {
      db.exec(sql)
      db.prepare('INSERT INTO schema_migrations (filename) VALUES (?)').run(file)
    })()
  }
}
```

Note: the `schema_migrations` table is created both in the migration file and in the runner itself (so the runner bootstraps correctly before running `0001_init.sql` which also contains it with `IF NOT EXISTS`).

```typescript
export async function hashPassword(plain: string): Promise<string> {
  return hash(plain)
}

export async function verifyPassword(plain: string, hashed: string): Promise<boolean> {
  return verify(hashed, plain)
}

// Pre-compute once at startup so timing attacks on unknown emails waste the same argon2 time
export async function getDummyHash(): Promise<string> {
  return hash(randomBytes(32).toString('hex'))
```

- [ ] **Step 2: Implement `src/auth/jwt.ts`**

```typescript
import {
  SignJWT,
  jwtVerify,
  createSecretKey,
  importPKCS8,
  importSPKI,
  exportJWK,
  calculateJwkThumbprint,
  type KeyLike,
} from 'jose'
import type { Config } from '../config.js'

export interface JwtPayload {
  sub: string
  email?: string
  role: 'user' | 'admin'
  scopes: string[]
  iss: string
  aud: string
  iat: number
  exp: number
  nbf: number
}

export interface JwkKey {
  kty: string
  n?: string
  e?: string
  kid: string
  alg: string
  use: string
}

export interface JwtSigner {
  sign(claims: Omit<JwtPayload, 'iss' | 'aud' | 'iat' | 'exp' | 'nbf'>): Promise<string>
  verify(token: string): Promise<JwtPayload>
  jwks?(): Promise<{ keys: JwkKey[] }>
}

export async function createJwtSigner(config: Config): Promise<JwtSigner> {
  if (config.jwtAlg === 'HS256') {
    const key = createSecretKey(Buffer.from(config.jwtSecret!))
    return {
      async sign(claims) {
        const now = Math.floor(Date.now() / 1000)
        return new SignJWT({
          email: claims.email,
          role: claims.role,
          scope: claims.scopes.join(' '),
        })
          .setProtectedHeader({ alg: 'HS256' })
          .setIssuer(config.jwtIssuer)
          .setAudience(config.jwtAudience)
          .setSubject(claims.sub)
          .setIssuedAt(now)
          .setNotBefore(now)
          .setExpirationTime(now + config.jwtTtlSeconds)
          .sign(key)
      },
      async verify(token) {
        const { payload } = await jwtVerify(token, key, {
          issuer: config.jwtIssuer,
          audience: config.jwtAudience,
          algorithms: ['HS256'],
        })
        return extractPayload(payload)
      },
    }
  }

  // RS256
  const privateKey = await importPKCS8(config.jwtPrivateKey!, 'RS256')
  const publicKey = await importSPKI(config.jwtPublicKey!, 'RS256')
  const jwkPublic = await exportJWK(publicKey)
  const kid = await calculateJwkThumbprint(jwkPublic)

  return {
    async sign(claims) {
      const now = Math.floor(Date.now() / 1000)
      return new SignJWT({
        email: claims.email,
        role: claims.role,
        scope: claims.scopes.join(' '),
      })
        .setProtectedHeader({ alg: 'RS256', kid })
        .setIssuer(config.jwtIssuer)
        .setAudience(config.jwtAudience)
        .setSubject(claims.sub)
        .setIssuedAt(now)
        .setNotBefore(now)
        .setExpirationTime(now + config.jwtTtlSeconds)
        .sign(privateKey)
    },
    async verify(token) {
      const { payload } = await jwtVerify(token, publicKey, {
        issuer: config.jwtIssuer,
        audience: config.jwtAudience,
        algorithms: ['RS256'],
      })
      return extractPayload(payload)
    },
    async jwks() {
      return {
        keys: [{ ...jwkPublic, kid, alg: 'RS256', use: 'sig' } as JwkKey],
      }
    },
  }
}

  return {
    sub: payload.sub as string,
    email: payload.email as string | undefined,
    role: (payload.role as 'user' | 'admin') ?? 'user',
    scopes: scope ? scope.split(' ') : [],
    iss: payload.iss as string,
    aud: Array.isArray(payload.aud) ? payload.aud[0] : (payload.aud as string),
    iat: payload.iat as number,
    exp: payload.exp as number,
    nbf: (payload.nbf as number) ?? (payload.iat as number),
```

```typescript
import type { FastifyInstance, FastifyReply, FastifyRequest, preHandlerHookHandler } from 'fastify'
import type { AppDeps } from '../app.js'
import type { Config } from '../config.js'
import { AppError } from '../errors.js'
import type { JwtPayload } from './jwt.js'

type AuthDeps = Pick<AppDeps, 'jwtSigner'>

export function makeAuthenticate(config: Config, deps: AuthDeps) {
  return async function authenticate(request: FastifyRequest, reply: FastifyReply): Promise<void> {
    const authHeader = request.headers.authorization
    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      void reply.header('WWW-Authenticate', 'Bearer')
      throw new AppError(401, 'unauthorized', 'missing or invalid authorization header')
    }

    const token = authHeader.slice(7)
    if (!token) {
      void reply.header('WWW-Authenticate', 'Bearer')
      throw new AppError(401, 'unauthorized', 'missing token')
    }

    try {
      const payload = await deps.jwtSigner.verify(token)
      request.user = payload
    } catch (err: unknown) {
      const message = err instanceof Error ? err.message : 'token validation failed'
      // IdP unavailable (only in OIDC mode)
      if (message.includes('idp_unavailable')) {
        throw new AppError(503, 'idp_unavailable', 'identity provider unavailable')
      }
      void reply.header('WWW-Authenticate', 'Bearer')
      throw new AppError(401, 'unauthorized', 'invalid or expired token')
    }
  }
}

export function requireScope(scope: string): preHandlerHookHandler {
  return async function (request: FastifyRequest, _reply: FastifyReply): Promise<void> {
    if (!request.user?.scopes.includes(scope)) {
      throw new AppError(403, 'forbidden', `missing required scope: ${scope}`)
    }
  }
}

export function requireRole(role: string): preHandlerHookHandler {
  return async function (request: FastifyRequest, _reply: FastifyReply): Promise<void> {
    if (request.user?.role !== role) {
      throw new AppError(403, 'forbidden', `missing required role: ${role}`)
    }
  }
}
```

```typescript
  _dummyHash = await getDummyHash()

function getDummy(): string {
  return _dummyHash ?? '$argon2id$v=19$m=65536,t=3,p=4$dummy$dummy'

export async function registerUser(
  db: Database.Database,
  email: string,
  password: string,
): Promise<{ id: string; email: string; role: 'user' | 'admin' }> {
  const normalised = email.toLowerCase()
  const existing = findUserByEmail(db, normalised)
  if (existing) throw new AppError(409, 'conflict', 'email already registered')

  const passwordHash = await hashPassword(password)
  const user = createUser(db, { email: normalised, passwordHash })
  return { id: user.id, email: user.email, role: user.role }
}

export async function loginUser(
  db: Database.Database,
  email: string,
  password: string,
): Promise<{ id: string; email: string; role: 'user' | 'admin' }> {
  const normalised = email.toLowerCase()
  const user = findUserByEmail(db, normalised)

  if (!user) {
    // timing: verify against dummy hash so response time is similar regardless of email existence
    await verifyPassword(password, getDummy()).catch(() => {})
    throw new AppError(401, 'unauthorized', 'invalid credentials')
  }

  const valid = await verifyPassword(password, user.password_hash)
  if (!valid) throw new AppError(401, 'unauthorized', 'invalid credentials')

  return { id: user.id, email: user.email, role: user.role }
}

function scopesForRole(role: 'user' | 'admin'): string[] {
  return role === 'admin'
    ? ['notes:read', 'notes:write', 'admin:read']
```

```typescript
    } else {
```

```typescript
      res.end('{}')

        .setAudience(audience)
        .setSubject(claims.sub)
        .setIssuedAt(now)
        .setNotBefore(now)
```

```typescript
import { createRemoteJWKSet, jwtVerify } from 'jose'
import type { Config } from '../config.js'
import type { JwtSigner, JwtPayload } from './jwt.js'

export function createOidcSigner(config: Config): JwtSigner {
  let jwks: ReturnType<typeof createRemoteJWKSet> | null = null

  function getJwks(): ReturnType<typeof createRemoteJWKSet> {
    if (!jwks) {
      const jwksUrl = config.oidcJwksUrl ?? `${config.jwtIssuer}/.well-known/openid-configuration`
      jwks = createRemoteJWKSet(new URL(jwksUrl))
    }
    return jwks
  }

  return {
    async sign() {
      throw new Error('OIDC mode: token issuance not supported')
    },

    async verify(token: string): Promise<JwtPayload> {
      let payload: Record<string, unknown>
      try {
        const result = await jwtVerify(token, getJwks(), {
          issuer: config.jwtIssuer,
          audience: config.jwtAudience,
          algorithms: ['RS256'],
        })
        payload = result.payload as Record<string, unknown>
      } catch (err: unknown) {
        const msg = err instanceof Error ? err.message : ''
        // Distinguish network/fetch errors (IdP down) from token errors
        if (
          msg.includes('ECONNREFUSED') ||
          msg.includes('ENOTFOUND') ||
          msg.includes('fetch') ||
          msg.includes('network')
        ) {
          const idpErr = new Error('idp_unavailable')
          idpErr.message = 'idp_unavailable'
          throw idpErr
        }
        throw err
      }

      const realmRoles =
        (payload.realm_access as { roles?: string[] } | undefined)?.roles ?? []
      const role: 'user' | 'admin' = realmRoles.includes('admin') ? 'admin' : 'user'
      const scope = typeof payload.scope === 'string' ? payload.scope : ''

      return {
        sub: payload.sub as string,
        email: payload.email as string | undefined,
        role,
        scopes: scope ? scope.split(' ') : [],
        iss: payload.iss as string,
        aud: Array.isArray(payload.aud) ? payload.aud[0] : (payload.aud as string),
        iat: payload.iat as number,
        exp: payload.exp as number,
        nbf: (payload.nbf as number) ?? (payload.iat as number),
      }
    },
  }
}
```

```typescript
import { loadConfig, ConfigError } from './config.js'
import { openDatabase } from './db/connection.js'
import { runMigrations } from './db/migrate.js'
import { buildApp } from './app.js'
import { createJwtSigner } from './auth/jwt.js'
import { createOidcSigner } from './auth/oidc.js'
import { createUser, findUserByEmail } from './services/users.js'
import { hashPassword } from './auth/password.js'
import { mkdirSync, dirname } from 'node:fs'
import { fileURLToPath } from 'node:url'
import { join } from 'node:path'

const __dirname = dirname(fileURLToPath(import.meta.url))
// In dev (src/main.ts): ../migrations = root/migrations
// In prod (dist/main.js): ../migrations = root/migrations (run from project root)
const migrationsDir = join(__dirname, '..', 'migrations')

async function main(): Promise<void> {
  // 1. Config
  let config
  try {
    config = loadConfig()
  } catch (err) {
    if (err instanceof ConfigError) {
      console.error(err.message)
      process.exit(1)
    }
    throw err
  }

  // 2. Database
  const dbDir = dirname(config.databasePath === ':memory:' ? '/tmp/x' : config.databasePath)
  if (config.databasePath !== ':memory:') {
    mkdirSync(dbDir, { recursive: true })
  }

  let db
  try {
    db = openDatabase(config.databasePath)
    runMigrations(db, migrationsDir)
  } catch (err) {
    console.error('Database startup failed:', err)
    process.exit(1)
  }

  // 3. Optional: seed admin
  if (config.seedAdminEmail && config.seedAdminPassword && config.authMode === 'local') {
    const existing = findUserByEmail(db, config.seedAdminEmail.toLowerCase())
    if (!existing) {
      const passwordHash = await hashPassword(config.seedAdminPassword)
      createUser(db, { email: config.seedAdminEmail.toLowerCase(), passwordHash, role: 'admin' })
    }
  }

  // 4. JWT signer
  const jwtSigner =
    config.authMode === 'oidc' ? createOidcSigner(config) : await createJwtSigner(config)

  // 5. App
  const shuttingDown = { value: false }
  const startedAt = new Date().toISOString()
  const app = await buildApp(config, {
    db,
    shuttingDown,
    startedAt,
    jwtSigner,
    authenticate: async () => {}, // replaced inside buildApp
  })

  // 6. Graceful shutdown
  let shuttingDownCount = 0
  const shutdown = async (signal: string): Promise<void> => {
    shuttingDownCount++
    if (shuttingDownCount > 1) {
      app.log.warn({ signal }, 'Forced shutdown')
      process.exit(1)
    }

    app.log.info({ signal, delaySeconds: config.shutdownDelaySeconds }, 'Graceful shutdown started')
    shuttingDown.value = true

    const safetyTimeout = setTimeout(() => {
      app.log.error('Shutdown safety timeout exceeded, forcing exit')
      process.exit(1)
    }, 25_000)
    safetyTimeout.unref()

    if (config.shutdownDelaySeconds > 0) {
      await new Promise((r) => setTimeout(r, config.shutdownDelaySeconds * 1000))
    }

    await app.close()
    db.close()
    clearTimeout(safetyTimeout)
    app.log.info('Shutdown complete')
    process.exit(0)
  }

  process.on('SIGTERM', () => void shutdown('SIGTERM'))
  process.on('SIGINT', () => void shutdown('SIGINT'))

  // 7. Listen
  try {
    await app.listen({ port: config.port, host: config.host })
  } catch (err) {
    app.log.error({ err }, 'Failed to start server')
    process.exit(1)
  }
}

void main()
```

```javascript
#!/usr/bin/env node
// scripts/smoke.mjs — no external dependencies

const BASE_URL = process.env.BASE_URL ?? 'http://localhost:8080'
const EXPECT_SHA = process.env.EXPECT_SHA
const EXPECT_ENV = process.env.EXPECT_ENV
const EXPECT_VERSION = process.env.EXPECT_VERSION
const SMOKE_TOKEN = process.env.SMOKE_TOKEN
const SMOKE_INSECURE = process.env.SMOKE_INSECURE === '1'
const TIMEOUT_MS = 5000
const READY_RETRIES = 30
const READY_INTERVAL_MS = 2000

let passed = 0
let failed = 0

function log(ok, label, detail = '') {
  const icon = ok ? '✓' : '✗'
  console.log(`${icon} ${label}${detail ? ` — ${detail}` : ''}`)
  if (ok) passed++
  else failed++
}

async function fetchWithTimeout(url, opts = {}) {
  const controller = new AbortController()
  const timer = setTimeout(() => controller.abort(), TIMEOUT_MS)
  try {
    const fetchOpts = { ...opts, signal: controller.signal }
    if (SMOKE_INSECURE && process.versions.node) {
      // Node 22: use --insecure flag or custom dispatcher
      // Simple approach: set env NODE_TLS_REJECT_UNAUTHORIZED=0 before running
    }
    return await fetch(url, fetchOpts)
  } finally {
    clearTimeout(timer)
  }
}

async function checkLive() {
  try {
    const res = await fetchWithTimeout(`${BASE_URL}/health/live`)
    const ok = res.status === 200
    log(ok, 'GET /health/live', `status=${res.status}`)
  } catch (err) {
    log(false, 'GET /health/live', err.message)
  }
}

async function checkReady() {
  for (let i = 0; i < READY_RETRIES; i++) {
    try {
      const res = await fetchWithTimeout(`${BASE_URL}/health/ready`)
      if (res.status === 200) {
        log(true, 'GET /health/ready', `status=200 (attempt ${i + 1})`)
        return
      }
    } catch {}
    if (i < READY_RETRIES - 1) {
      await new Promise((r) => setTimeout(r, READY_INTERVAL_MS))
    }
  }
  log(false, 'GET /health/ready', `not ready after ${READY_RETRIES} attempts`)
}

async function checkVersion() {
  try {
    const res = await fetchWithTimeout(`${BASE_URL}/version`)
    const body = await res.json()
    const ok = res.status === 200
    log(ok, 'GET /version', `status=${res.status}`)

    if (EXPECT_SHA) {
      const match = body.git_sha === EXPECT_SHA
      log(match, '  git_sha matches', `expected=${EXPECT_SHA} got=${body.git_sha}`)
    }
    if (EXPECT_ENV) {
      const match = body.environment === EXPECT_ENV
      log(match, '  environment matches', `expected=${EXPECT_ENV} got=${body.environment}`)
    }
    if (EXPECT_VERSION) {
      const match = body.version === EXPECT_VERSION
      log(match, '  version matches', `expected=${EXPECT_VERSION} got=${body.version}`)
    }
  } catch (err) {
    log(false, 'GET /version', err.message)
  }
}

async function checkMeNoToken() {
  try {
    const res = await fetchWithTimeout(`${BASE_URL}/me`)
    const ok = res.status === 401
    log(ok, 'GET /me (no token) → 401', `status=${res.status}`)
  } catch (err) {
    log(false, 'GET /me (no token)', err.message)
  }
}

async function checkMeWithToken() {
  if (!SMOKE_TOKEN) return
  try {
    const res = await fetchWithTimeout(`${BASE_URL}/me`, {
      headers: { Authorization: `Bearer ${SMOKE_TOKEN}` },
    })
    const ok = res.status === 200
    log(ok, 'GET /me (with SMOKE_TOKEN) → 200', `status=${res.status}`)
  } catch (err) {
    log(false, 'GET /me (with SMOKE_TOKEN)', err.message)
  }
}

async function main() {
  if (SMOKE_INSECURE) process.env.NODE_TLS_REJECT_UNAUTHORIZED = '0'

  await checkReady()
  await checkVersion()
  await checkMeNoToken()

  console.log(`\n${passed} passed, ${failed} failed`)
  process.exit(failed > 0 ? 1 : 0)
}

main().catch((err) => {
  console.error('Smoke test error:', err)
  process.exit(1)
})
```

## Why SQLite limits replicas to 1

SQLite is a file-based database. WAL mode allows concurrent reads, but only one writer at a time — and the file must live on a single node's filesystem. Running 2+ replicas against the same volume would cause locking issues. The natural migration path is PostgreSQL (via `pg` or `postgres` npm packages with the same service layer).
